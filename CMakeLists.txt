cmake_minimum_required(VERSION 3.10)

if (${CMAKE_SOURCE_DIR} STREQUAL ${CMAKE_BINARY_DIR})
    message(FATAL_ERROR
        "In-source builds not allowed.
        Please make a new directory (called a build directory) and run CMake from there.
        You may need to remove CMakeCache.txt."
    )
endif ()

project(wowdecomp CXX C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if (NOT CMAKE_SYSTEM_NAME MATCHES "Windows")
    message(FATAL_ERROR "wowdecomp targets Windows only.")
endif ()

# x64 is the target. Win32 stays buildable: it is the configuration whose struct layouts can be
# checked against Wow.exe 3.3.5a (12340).
if (CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(WOWDECOMP_ARCH_64 1)
    add_compile_definitions(WOWDECOMP_ARCH_64=1)
elseif (CMAKE_SIZEOF_VOID_P EQUAL 4)
    set(WOWDECOMP_ARCH_32 1)
    add_compile_definitions(WOWDECOMP_ARCH_32=1)
    # x87 floating point like the VS2005 original; keeps Win32 code comparable with the binary (re/tools/verify.py).
    if (MSVC)
        add_compile_options(/arch:IA32 /fp:fast)
    endif ()
    # 32-bit time_t like the original build: time/localtime/mktime resolve to _time32/_localtime32/_mktime32.
    add_compile_definitions(_USE_32BIT_TIME_T)
else ()
    message(FATAL_ERROR "Unsupported pointer size.")
endif ()

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

add_compile_definitions(
    WIN32
    _WINDOWS
    NOMINMAX
    WIN32_LEAN_AND_MEAN
    _CRT_SECURE_NO_WARNINGS
)

# Off by default: the client checks MPQ file MD5s (signaturefile) and GlueXML/FrameXML/AddOn .toc.sig files like the
# original. ON accepts unsigned or modified game data and interface files.
option(WOWDECOMP_DISABLE_SIGNATURE_CHECKS "Skip MPQ file signature and .toc signature checks" OFF)
if (WOWDECOMP_DISABLE_SIGNATURE_CHECKS)
    add_compile_definitions(DISABLE_SIGNATURE_CHECKS=1)
endif ()

# Reads modern-client content the stock client cannot: .db2 tables (WDC1/2/3), FileDataID -> path
# resolution, and later the modern M2 model format. Off by default so the stock 3.3.5a build is
# byte-for-byte the code it was before. See Engine/Source/ModernAssets/README.md.
option(WOWDECOMP_MODERN_ASSETS "Read modern-client DB2 tables and modern M2 models" OFF)
if (WOWDECOMP_MODERN_ASSETS)
    add_compile_definitions(WOWDECOMP_MODERN_ASSETS=1)
endif ()

# Third-party code is linked, not decompiled, for now.
if (EXISTS ${CMAKE_SOURCE_DIR}/ThirdParty/CMakeLists.txt)
    add_subdirectory(ThirdParty)
endif ()

# NOT IN BINARY, build-only. Full PDBs so the in-tree crash logger can turn a fault address into a named
# frame. Tools/WowClient/crash_log.cpp already walks the stack with SymFromAddr and SymGetLineFromAddr64,
# but SymInitialize finds nothing without a PDB, which is why every crash so far reports only "module+0xNNN".
# add_compile_options/add_link_options append to the existing flags, so the CRT defaults and /MD stay intact
# (assigning CMAKE_CXX_FLAGS_RELEASE directly does not, and breaks XvidDecoder's linkage).
option(WOWDECOMP_DEBUG_SYMBOLS "Build with PDBs and /Zi so in-tree crash stacks can be symbolised" OFF)
if (WOWDECOMP_DEBUG_SYMBOLS)
    add_compile_options(/Zi)
    add_link_options(/DEBUG /DEBUGTYPE:CV)
endif ()
